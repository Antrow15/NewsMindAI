"""Persist and read cleaning quality reports from DB."""

from __future__ import annotations

from typing import Any

from dao.cleaning_report_dao import CleaningReportDAO
from utils.errors import NotFoundError
from utils.validators import require_positive_id


class ReportService:
    def __init__(self) -> None:
        self.dao = CleaningReportDAO()

    def save_cleaning_report(
        self,
        *,
        report_type: str,
        result: dict[str, Any],
        source_files: list[str] | str | None = None,
        created_by: int | None = None,
    ) -> int:
        if isinstance(source_files, list):
            source_text = ", ".join(source_files)
        else:
            source_text = source_files or ""
        stats = result.get("stats") or {}
        report = result.get("report") or {}
        return self.dao.create(
            report_type=report_type,
            report=report,
            input_count=int(stats.get("input") or 0),
            kept_count=int(stats.get("kept") or 0),
            rejected_count=int(stats.get("rejected") or 0),
            source_files=source_text,
            created_by=created_by,
        )

    def list_recent(self, limit: int = 15) -> list[dict[str, Any]]:
        return self.dao.list_recent(limit=limit)

    def get(self, report_id: int) -> dict[str, Any]:
        row = self.dao.read(require_positive_id(report_id, "Report ID"))
        if not row:
            raise NotFoundError(f"Cleaning report ID {report_id} was not found.")
        return row

# Learning guide: `services/user/`

## Why this folder exists

User lifecycle operations — **creating accounts** and **administrator password reset** — need validation, repository coordination, and authorization in one place. The console could call DAOs from `app.py`, but then CLI and tests would duplicate employee+user+desk creation rules.

`UserService` is the **domain service for accounts**. It does not handle login sessions (`authenticate` lives in `modules.authentication`). It does not paint the terminal (`app.do_add_user` collects prompts).

Password policy reminder:

- **Add user** is admin-only (menu gated by `CAP_ADMIN` / admin role in `app._build_menu`).
- **Reset password** is admin-only inside the service via `require_capability_session(CAP_ADMIN)`.
- There is **no** forgot-password API here or in authentication.

---

## How this package connects

```
app.do_add_user  →  UserService.add_user
app.do_admin_reset_password  →  UserService.reset_password

UserService
  → UserRepository (employee, user, role, desk)
  → hash_password (security)
  → admin_reset_password (auth_service)
  → require_capability_session (only on reset)
```

Login still uses `UserDAO` directly from `auth_service` for read-by-username — that is intentional (auth is a separate bounded context).

---

## `user_service.py` — class `UserService`

**Constructor:** optional `UserRepository` injection for tests.

### `add_user(...)`

Parameters: `username`, `password`, `first_name`, `last_name`, `email`, `role_code` (default `journalist`), optional `employee_code`.

Flow in prose:

1. Run validators: `require_username`, `require_password`, names, `require_email`, `require_role_code`.
2. Reject duplicate usernames via `repo.find_by_username`.
3. Resolve `role_id` from `repo.role_id_by_code`; unknown role → `ValidationError`.
4. Build `Employee` with generated or supplied `employee_code`, department `Newsroom`, designation = role code.
5. `repo.create_employee` → `emp_id`.
6. Build `User` with `hash_password(password)` — plaintext never touches the DAO.
7. `repo.create_user` → `user_id`.
8. `repo.ensure_desk(user_id, role_code)` so journalist gets a journalist profile row, etc.
9. Return `{"user_id", "username", "role_code"}` for `app.py` to print.

### `find_user(username)`

Validates username, returns row or raises `NotFoundError`. Utility for future admin tooling/tests.

### `reset_password(username, new_password)`

**Admin-only:** `require_capability_session(CAP_ADMIN)` then delegates to `admin_reset_password(username, new_password)` in auth module, which updates hash via `UserDAO`.

Non-admin callers get `PermissionDeniedError` (see `tests/test_password_flows.py`).

---

## `__init__.py`

Exports `UserService` for `from services.user import UserService` in `app.py`.

---

## Relationship to authentication

| Action | Where it lives |
|--------|----------------|
| Login / logout / session | `modules.authentication` |
| Change own password | `change_password` in auth_service; menu in `app.do_change_password` |
| Admin reset another user | `UserService.reset_password` → `admin_reset_password` |
| Hash algorithm | `modules.authentication.security` |

Keeping admin reset in auth_service as a function while `UserService` enforces capability avoids circular imports and keeps PBKDF2 + DAO update in one implementation.

"""User domain service — employee/user/desk creation with validation."""

from __future__ import annotations

from typing import Any

from config.constants import CAP_ADMIN
from entities.employee import Employee
from entities.user import User
from modules.authentication.auth_service import admin_reset_password
from modules.authentication.security import hash_password
from repository.user_repository import UserRepository
from services.access import require_capability_session
from utils.errors import NotFoundError, ValidationError
from utils.validators import (
    require_email,
    require_nonempty,
    require_password,
    require_role_code,
    require_username,
)


class UserService:
    def __init__(self, repo: UserRepository | None = None) -> None:
        self.repo = repo or UserRepository()

    def add_user(
        self,
        *,
        username: str,
        password: str,
        first_name: str,
        last_name: str,
        email: str,
        role_code: str = "journalist",
        employee_code: str | None = None,
    ) -> dict[str, Any]:
        username = require_username(username)
        password = require_password(password)
        first_name = require_nonempty(first_name, "First name")
        last_name = require_nonempty(last_name, "Last name")
        email = require_email(email)
        role_code = require_role_code(role_code)

        if self.repo.find_by_username(username):
            raise ValidationError(f"Username '{username}' already exists.")

        role_id = self.repo.role_id_by_code(role_code)
        if role_id is None:
            raise ValidationError(f"Unknown role code: {role_code}")

        code = (employee_code or f"EMP-{username.upper()}").strip()
        emp_id = self.repo.create_employee(
            Employee(
                employee_code=code,
                first_name=first_name,
                last_name=last_name,
                email=email,
                department="Newsroom",
                designation=role_code,
            )
        )
        user_id = self.repo.create_user(
            User(
                employee_id=emp_id,
                role_id=role_id,
                username=username,
                password_hash=hash_password(password),
                account_status="active",
            )
        )
        self.repo.ensure_desk(user_id, role_code)
        return {"user_id": user_id, "username": username, "role_code": role_code}

    def find_user(self, username: str) -> dict[str, Any]:
        username = require_username(username)
        user = self.repo.find_by_username(username)
        if not user:
            raise NotFoundError(f"User '{username}' was not found.")
        return user

    def reset_password(self, username: str, new_password: str) -> dict[str, Any]:
        """Admin-only password reset for another account."""
        require_capability_session(CAP_ADMIN)
        return admin_reset_password(username, new_password)

from __future__ import annotations

import sys
from pathlib import Path

import pytest

ROOT = Path(__file__).resolve().parents[1]
PARENT = ROOT.parent.resolve()

cleaned: list[str] = []
for entry in sys.path:
    try:
        resolved = Path(entry).resolve()

        cleaned.append(entry)
        continue
    if resolved == PARENT:

sys.path.insert(0, str(ROOT))

"""Human-readable console formatting (no JSON dumps)."""

from __future__ import annotations

from typing import Any, Mapping, Optional, Sequence


def line(char: str = "=", width: int = 56) -> str:
    return char * width


def header(title: str) -> None:
    print()
    print(line("="))
    print(f"  {title}")
    print(line("="))


def success(message: str) -> None:
    print(f"[OK] {message}")


def error(message: str) -> None:
    print(f"[ERROR] {message}")


def info(message: str) -> None:
    print(f"[INFO] {message}")


def kv(label: str, value: Any) -> None:
    text = "-" if value is None or value == "" else value
    print(f"  {label} - {text}")


def blank() -> None:
    print()


def show_user(user: Mapping[str, Any]) -> None:
    header("Session")
    kv("Username", user.get("username"))
    kv("Full Name", user.get("full_name"))
    kv("Role", user.get("role_name") or user.get("role_code"))
    kv("Role Code", user.get("role_code"))
    caps = user.get("capabilities") or []
    kv("Capabilities", ", ".join(caps) if caps else "none")
    profiles = user.get("profiles") or {}
    if profiles:
        kv("Profiles", ", ".join(f"{k}={v}" for k, v in profiles.items()))
    blank()


def show_dict_rows(title: str, rows: Sequence[Mapping[str, Any]], fields: Sequence[tuple[str, str]]) -> None:
    """fields: list of (label, key)."""
    header(title)
    if not rows:
        info("No records found.")
        blank()
        return
    for i, row in enumerate(rows, start=1):
        print(f"  --- Record {i} ---")
        for label, key in fields:
            kv(label, row.get(key))
        print()
    kv("Total", len(rows))
    blank()


def show_stats(title: str, stats: Mapping[str, Any]) -> None:
    header(title)
    for key, value in stats.items():
        label = str(key).replace("_", " ").title()
        if isinstance(value, Mapping):
            kv(label, ", ".join(f"{k}={v}" for k, v in value.items()))
        else:
            kv(label, value)
    blank()


CANCEL_WORDS = {"q", "quit", "exit", "cancel"}


class UserCancelled(Exception):
    """User asked to cancel the current prompt/action."""


def is_cancel(value: str) -> bool:
    return (value or "").strip().lower() in CANCEL_WORDS


def prompt_nonempty(label: str) -> str:
    while True:
        value = input(f"{label}: ").strip()
        if is_cancel(value):
            raise UserCancelled("Cancelled.")
        if value:
            return value
        error(f"{label} cannot be empty.")


def prompt_optional(label: str, default: str = "") -> str:
    value = input(f"{label}: ").strip()
    if is_cancel(value):
        raise UserCancelled("Cancelled.")
    return value if value else default


def prompt_choice(label: str, allowed: Sequence[str], *, allow_empty: bool = False) -> Optional[str]:
    options = ", ".join(allowed)
    while True:
        value = input(f"{label} [{options}]: ").strip().lower()
        if is_cancel(value):
            raise UserCancelled("Cancelled.")
        if not value:
            if allow_empty:
                return None
            error(f"{label} is required. Choose one of: {options}")
            continue
        if value in {a.lower() for a in allowed}:
            return value
        error(f"Invalid {label.lower()}. Allowed: {options}")

"""Normalize any CSV news-like columns into canonical fields.

Works for current samples and unknown future datasets (CSV only).
"""

from __future__ import annotations

from typing import Any

from utils.helpers import coerce_str

TITLE_ALIASES = (
    "title",
    "headline",
    "headline_text",
    "headlines",
    "news_title",
    "article_title",
    "name",
    "heading",
    "primary_heading",
    "story_title",
    "subject_line",
)
TEXT_ALIASES = (
    "text",
    "content",
    "content_body",
    "body",
    "article",
    "article_body",
    "description",
    "short_description",
    "summary",
    "story",
    "full_text",
    "full_article",
    "article_text",
    "news_text",
    "content_text",
)
CATEGORY_ALIASES = (
    "category",
    "category_name",
    "subject",
    "section",
    "topic",
    "topic_label",
    "tags",
    "label",
    "class",
    "genre",
    "channel",
)
DATE_ALIASES = (
    "date",
    "published_date",
    "publish_date",
    "publication_date",
    "datetime",
    "time",
    "created_at",
    "published_at",
)
STATUS_ALIASES = ("status", "state")
KIND_ALIASES = ("content_kind", "kind", "type", "article_type", "news_type")
SOURCE_ALIASES = ("source_dataset", "source", "dataset", "origin")
SUMMARY_ALIASES = ("summary", "excerpt", "teaser", "abstract")
SKIP_KEYS = {"id", "index", "unnamed: 0", "row_id", "uuid"}


def _first(row: dict[str, Any], aliases: tuple[str, ...]) -> str:
    for key in aliases:
        if key in row and coerce_str(row.get(key)):
            return coerce_str(row.get(key))
    return ""


def _longest_text_value(row: dict[str, Any], *, exclude: set[str] | None = None) -> str:
    exclude = exclude or set()
    best = ""
    for key, value in row.items():
        if key in exclude or key in SKIP_KEYS:
            continue
        text = coerce_str(value)
        if len(text) > len(best):
            best = text
    return best


def normalize_row(row: dict[str, Any], *, source_file: str = "") -> dict[str, Any]:
    """Map arbitrary CSV columns → title/text/category/date/..."""
    raw = {
        str(k).strip().lower(): v
        for k, v in row.items()
        if str(k).strip() and not str(k).lower().startswith("unnamed")
    }
    raw.pop("", None)

    title = _first(raw, TITLE_ALIASES)
    text = _first(raw, TEXT_ALIASES)
    category = (_first(raw, CATEGORY_ALIASES) or "General")[:128]
    if category.count(",") >= 2:
        category = category.split(",", 1)[0].strip()[:128] or "General"
    date_val = _first(raw, DATE_ALIASES)[:64]
    status = (_first(raw, STATUS_ALIASES) or "draft").lower()
    kind = (_first(raw, KIND_ALIASES) or "article").lower()
    if kind not in {"article", "news"}:
        kind = "article"
    source = (_first(raw, SOURCE_ALIASES) or source_file)[:128]
    summary = _first(raw, SUMMARY_ALIASES)

    # Unknown dataset fallback: use longest string field as body/title
    if not text:
        text = _longest_text_value(raw, exclude={*(TITLE_ALIASES), *(CATEGORY_ALIASES), *(DATE_ALIASES)})
    if not title and text:
        title = text[:120].strip()
    if title and not text:
        text = title
        if category.lower() in {"", "general", "unknown"}:
            category = "Headline"

    out = dict(raw)
    out.update(
        {
            "title": title,
            "text": text,
            "content_body": text,
            "category": category,
            "category_name": category,
            "date": date_val,
            "status": status if status else "draft",
            "content_kind": kind,
            "summary": summary,
            "source_dataset": source,
            "source_type": f"csv:{source_file}" if source_file else coerce_str(raw.get("source_type")),
            "_raw_title": title,
            "_raw_text": text,
            "_source_file": source_file,
        }
    )
    return out

from __future__ import annotations

import re
from pathlib import Path
from typing import Any, Iterable


ROOT = Path(__file__).resolve().parents[1]


def project_path(*parts: str) -> Path:
    return ROOT.joinpath(*parts)


def coerce_str(value: Any, default: str = "") -> str:
    if value is None:
        return default
    return str(value).strip()


def normalize_key(text: str) -> str:
    text = coerce_str(text).lower()
    text = re.sub(r"<[^>]+>", " ", text)
    text = re.sub(r"[^a-z0-9\s]", " ", text)
    text = re.sub(r"\s+", " ", text).strip()
    return text


def chunked(items: Iterable[Any], size: int) -> list[list[Any]]:
    buf: list[Any] = []
    out: list[list[Any]] = []
    for item in items:
        buf.append(item)
        if len(buf) >= size:
            out.append(buf)
            buf = []
    if buf:
        out.append(buf)
    return out

from __future__ import annotations

import logging
from pathlib import Path

from utils.helpers import project_path


def get_logger(name: str = "newsmind") -> logging.Logger:
    logger = logging.getLogger(name)
    if logger.handlers:
        return logger
    logger.setLevel(logging.INFO)
    fmt = logging.Formatter("%(asctime)s | %(levelname)s | %(name)s | %(message)s")

    log_dir = project_path("logs")
    log_dir.mkdir(parents=True, exist_ok=True)
    file_handler = logging.FileHandler(log_dir / "application.log", encoding="utf-8")
    file_handler.setFormatter(fmt)
    file_handler.setLevel(logging.INFO)
    # Console: show INFO+ so pipeline steps are visible in the terminal
    stream_handler = logging.StreamHandler()
    stream_handler.setFormatter(fmt)
    stream_handler.setLevel(logging.INFO)
    logger.addHandler(file_handler)
    logger.addHandler(stream_handler)
    logger.propagate = False
    return logger

        log_dir.mkdir(parents=True, exist_ok=True)
        handler = logging.FileHandler(log_dir / "application.log", encoding="utf-8")
        handler.setFormatter(
            logging.Formatter("%(asctime)s | %(levelname)s | %(name)s | %(message)s")
        )
        logger.addHandler(handler)
        logger.propagate = False
    _file_log = logger
    return logger


def progress(message: str) -> None:
    """Show a clear step on the console and mirror it to application.log."""
    text = str(message).strip()
    if not text:
        return
    info(text)
    _file_logger().info("%s", text)

# Learning guide: `utils/`

## Why this folder exists

Application code repeats small concerns: logging, terminal formatting, validation errors, CSV column normalization, and path helpers. If every service inlined those, rules would drift — one path might allow a two-character password while another would not.

`utils/` is the **shared kernel** for cross-cutting helpers. It imports `config.constants` for allow-lists (statuses, role codes, password rules) but does not import DAOs or services, so low-level modules can depend on it without cycles.

---

## How other packages use `utils/`

| Consumer | Typical imports |
|----------|-----------------|
| `app.py` | `console_ui`, `logger`, `validators`, `errors` |
| All DAOs | `validators`, `errors` (`ValidationError`, `NotFoundError`) |
| `services.*` | `validators`, `errors`, `helpers`, `csv_fields` |
| `cleaners/stages` | `validators.validate_content_record`, `helpers.coerce_str` |
| `modules.authentication` | `validators.require_password`, `project_path` |

`services.access` maps `AuthenticationError` to `AuthError` but the message strings often originate in validators.

---

## File-by-file walkthrough

### `errors.py`

Defines the exception hierarchy for the console:

- **`AppError`** — base with `.message` for `_handle` in `app.py`.
- **`ValidationError`** — bad user input before DB write.
- **`NotFoundError`** — missing entity.
- **`AuthError`** — not logged in or bad session (from access layer).
- **`PermissionDeniedError`** — logged in but missing capability.

Typed errors let the console print user-friendly text instead of stack traces for expected failures.

### `validators.py`

The largest file. Functions fall into groups:

- **Presence and shape:** `require_nonempty`, `require_optional_str`, `require_positive_id`, `require_username`, `require_email`, `require_password` (length/complexity from config).
- **Enumerations:** `require_content_status`, `require_pitch_status`, `require_account_status`, `require_role_code`, `require_severity`, etc. — each checks membership in constants or local tuples.
- **Content pipeline:** `require_title`, `require_body`, `validate_content_record` — used by cleaners, persistence, and desk CRUD so the same rules apply to CSV rows and manual entry.

**`require_password`** is central to password policy: admin add-user, change-password, and admin reset all funnel new secrets through it.

### `logger.py`

**`get_logger(name)`** returns a module-scoped logger writing to `logs/application.log` with a consistent format. Packages use names like `"app"`, `"db"`, `"ingestion"`, `"cleaning"` so you can filter log noise.

### `console_ui.py`

Human-facing terminal UX:

- **`header`, `success`, `error`, `info`, `kv`, `blank`** — consistent output.
- **`prompt_nonempty`, `prompt_optional`, `prompt_choice`** — input loops with validation.
- **`is_cancel` / `UserCancelled`** — `q` and `exit` abort a step without killing the app.
- **`show_dict_rows`, `show_stats`, `show_user`** — tabular display for lists and login banner.

`app.py` is mostly menu wiring; readability comes from these helpers.

### `csv_fields.py`

Normalizes heterogeneous CSV column names into the dict shape ingestion and cleaners expect (`title`, `text`, `category`, etc.). Ingestion calls `normalize_row` so one dataset with `headline` and another with `title` converge.

### `helpers.py`

**`project_path(*parts)`** — resolves paths under the `New folder` root (session file, data dirs). **`coerce_str`** and similar small coercions support cleaners without pulling pandas into every stage.

---

## Keyword: validation at boundaries

NewsMind validates at **service and DAO** boundaries, not only in `app.py` CLI prompts. That matters when:

- Tests call services directly without the console.
- Pipeline code inserts thousands of rows — `validate_content_record` rejects bad rows before SQL.

When you add a field, add a `require_*` or extend `validate_content_record` once, then call it from cleaner, persistence, and desk paths.

"""Field validation used by DAOs, services, and cleaners."""

from __future__ import annotations

import re
from typing import Any, Optional

from config.constants import (

from utils.errors import ValidationError

EMAIL_RE = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
ALLOWED_SEVERITIES = ("low", "medium", "high", "critical")
ALLOWED_PRIORITIES = ("low", "medium", "high", "urgent")
ALLOWED_ARTICLE_TYPES = ("feature", "analysis", "opinion", "profile", "explainer")
ALLOWED_DESK_STATUSES = ("active", "inactive")
ALLOWED_AVAILABILITY = ("available", "busy", "away", "leave")
ALLOWED_EMPLOYMENT = ("active", "inactive", "terminated")
ALLOWED_CLEANING_STATUS = ("pending", "kept", "rejected", "ingested")


def is_nonempty(value: Any) -> bool:
    return value is not None and str(value).strip() != ""


def require_nonempty(value: Any, label: str) -> str:
    text = "" if value is None else str(value).strip()
    if not text:
        raise ValidationError(f"{label} cannot be empty.")
    return text


def require_optional_str(value: Any, label: str, *, max_len: Optional[int] = None) -> Optional[str]:
    if value is None:
        return None
    text = str(value).strip()
    if not text:
        return None
    if max_len is not None and len(text) > max_len:
        raise ValidationError(f"{label} must be at most {max_len} characters.")
    return text


def is_valid_email(value: str) -> bool:
    return bool(EMAIL_RE.match((value or "").strip()))


def require_email(value: Any) -> str:
    email = require_nonempty(value, "Email")
    if not is_valid_email(email):
        raise ValidationError("A valid email is required.")
    return email


def is_valid_title(value: str) -> bool:
    return len((value or "").strip()) >= MIN_TITLE_LENGTH


def require_title(value: Any) -> str:
    title = require_nonempty(value, "Title")
    if len(title) < MIN_TITLE_LENGTH:
        raise ValidationError(f"Title must be at least {MIN_TITLE_LENGTH} characters.")
    if len(title) > 512:
        raise ValidationError("Title must be at most 512 characters.")
    return title


def is_valid_body(value: str, *, headline_ok: bool = False) -> bool:
    text = (value or "").strip()
    if len(text) >= MIN_BODY_LENGTH:
        return True
    if headline_ok and len(text) >= MIN_TITLE_LENGTH:
        return True
    return False


def require_body(value: Any, *, headline_ok: bool = False) -> str:
    body = require_nonempty(value, "Body")
    if not is_valid_body(body, headline_ok=headline_ok):
        raise ValidationError(f"Body must be at least {MIN_BODY_LENGTH} characters.")
    return body


def require_choice(value: Any, label: str, allowed: tuple[str, ...]) -> str:
    text = require_nonempty(value, label).lower()
    if text not in {a.lower() for a in allowed}:
        raise ValidationError(f"Invalid {label.lower()} '{value}'. Allowed: {', '.join(allowed)}")
    return text


def is_allowed_content_status(value: str) -> bool:
    return (value or "").strip().lower() in ALLOWED_CONTENT_STATUSES


def is_allowed_pitch_status(value: str) -> bool:
    return (value or "").strip().lower() in ALLOWED_PITCH_STATUSES


def is_allowed_account_status(value: str) -> bool:
    return (value or "").strip().lower() in ALLOWED_ACCOUNT_STATUSES


def is_allowed_content_kind(value: str) -> bool:
    return (value or "").strip().lower() in CONTENT_KINDS


def require_content_status(value: str) -> str:
    return require_choice(value if value is not None else "", "Status", ALLOWED_CONTENT_STATUSES)


def require_pitch_status(value: str) -> str:
    return require_choice(value if value is not None else "", "Status", ALLOWED_PITCH_STATUSES)


def require_account_status(value: str) -> str:
    return require_choice(value if value is not None else "", "Account status", ALLOWED_ACCOUNT_STATUSES)


def require_content_kind(value: str) -> str:
    return require_choice(value if value is not None else "", "Content kind", CONTENT_KINDS)


def require_severity(value: Any) -> str:
    return require_choice(value if value is not None else "", "Severity", ALLOWED_SEVERITIES)


def require_priority(value: Any) -> str:
    return require_choice(value if value is not None else "", "Priority", ALLOWED_PRIORITIES)


def require_article_type(value: Any) -> str:
    return require_choice(value if value is not None else "", "Article type", ALLOWED_ARTICLE_TYPES)


def require_desk_status(value: Any) -> str:
    return require_choice(value if value is not None else "", "Status", ALLOWED_DESK_STATUSES)


def require_availability(value: Any) -> str:
    return require_choice(value if value is not None else "", "Availability", ALLOWED_AVAILABILITY)


def require_employment_status(value: Any) -> str:
    return require_choice(value if value is not None else "", "Employment status", ALLOWED_EMPLOYMENT)


def require_cleaning_status(value: Any) -> str:
    return require_choice(value if value is not None else "", "Cleaning status", ALLOWED_CLEANING_STATUS)


def require_positive_id(raw: Any, label: str = "ID") -> int:
    if isinstance(raw, int):
        value = raw
    else:
        text = "" if raw is None else str(raw).strip()
        if not text:
            raise ValidationError(f"{label} cannot be empty.")
        try:
            value = int(text)
        except ValueError as exc:
            raise ValidationError(f"{label} must be a number.") from exc
    if value <= 0:
        raise ValidationError(f"{label} must be a positive number.")
    return value


def require_optional_positive_id(raw: Any, label: str = "ID") -> Optional[int]:
    if raw is None or str(raw).strip() == "":
        return None
    return require_positive_id(raw, label)


def require_username(value: Any) -> str:
    username = require_nonempty(value, "Username")
    if len(username) < 3:
        raise ValidationError("Username must be at least 3 characters.")
    if " " in username:
        raise ValidationError("Username cannot contain spaces.")
    return username


_PASSWORD_RULES = (
    (r".{10,}", "at least 10 characters"),
    (r"[A-Z]", "one uppercase letter"),
    (r"[a-z]", "one lowercase letter"),
    (r"\d", "one digit"),
    (r"[^A-Za-z0-9]", "one special character"),
)


def require_password(value: Any) -> str:
    """Strong password policy for change/reset (not applied to login of existing users)."""
    password = "" if value is None else str(value)
    if not password:
        raise ValidationError("Password cannot be empty.")
    if password.strip() != password:
        raise ValidationError("Password cannot start or end with spaces.")
    missing = [label for pattern, label in _PASSWORD_RULES if not re.search(pattern, password)]
    if missing:
        raise ValidationError(
            "Password must include: " + ", ".join(missing) + "."
        )
    if len(password) > 128:
        raise ValidationError("Password must be at most 128 characters.")
    return password


def require_role_code(value: Any) -> str:
    code = require_nonempty(value, "Role code").lower()
    if " " in code:
        raise ValidationError("Role code cannot contain spaces.")
    return code


def require_category_name(value: Any) -> str:
    name = require_nonempty(value, "Category name")
    if len(name) > 128:
        raise ValidationError("Category name must be at most 128 characters.")
    return name


def require_score(value: Any, label: str) -> float:
    try:
        score = float(value if value is not None else 0)
    except (TypeError, ValueError) as exc:
        raise ValidationError(f"{label} must be a number.") from exc
    if score < 0 or score > 1_000:
        raise ValidationError(f"{label} must be between 0 and 1000.")
    return score


def require_read_time(value: Any) -> Optional[int]:
    if value is None or str(value).strip() == "":
        return None
    try:
        minutes = int(value)
    except (TypeError, ValueError) as exc:
        raise ValidationError("Estimated read time must be a whole number.") from exc
    if minutes < 1 or minutes > 240:
        raise ValidationError("Estimated read time must be between 1 and 240 minutes.")
    return minutes


def validate_content_record(record: dict[str, Any]) -> list[str]:
    errors: list[str] = []
    title = record.get("title", "")
    body = record.get("content_body") or record.get("text") or ""
    category = str(record.get("category") or record.get("category_name") or "").lower()
    headline_ok = category in {"headline", "headlines", "sarcasm"} or (
        str(title).strip() == str(body).strip() and len(str(title).strip()) >= MIN_TITLE_LENGTH
    )
    if not is_valid_title(title):
        errors.append("title_too_short")
    if not is_valid_body(body, headline_ok=headline_ok):
        errors.append("body_too_short")
    status = (record.get("status") or "draft").lower()
    if not is_allowed_content_status(status):
        errors.append("invalid_status")
    return errors
