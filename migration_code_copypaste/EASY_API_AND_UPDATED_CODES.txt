    function fieldHtml(f) {
      const val = f.value ?? "";
      if (f.type === "textarea") {
        return `<label>${f.label}<textarea data-key="${f.key}" data-in="${f.in || "body"}">${escapeHtml(String(val))}</textarea></label>`;
      }
      return `<label>${f.label}<input data-key="${f.key}" data-in="${f.in || "body"}" type="${f.type === "number" ? "number" : "text"}" value="${escapeAttr(String(val))}" /></label>`;
    }

    function escapeHtml(s) {
      return s.replace(/&/g, "&amp;").replace(/</g, "&lt;");

    function readFields(ep) {
      const pathVals = {};
      const query = {};
      const body = {};
      $("formPanel").querySelectorAll("[data-key]").forEach((el) => {
        const key = el.dataset.key;
        const where = el.dataset.in || "body";
        let val = el.value;
        if (el.type === "number" && val !== "") val = Number(val);
        if (where === "path") pathVals[key] = val;
        else if (where === "query") {
          if (val !== "" && val !== null && val !== undefined) query[key] = val;
        } else if (val !== "" || key === "text") {
          body[key] = val;
        }
      });

      if (body.documents_json !== undefined) {
        try { body.documents = JSON.parse(body.documents_json || "[]"); }
        catch { throw new Error("Documents JSON is invalid"); }
        delete body.documents_json;
      }
      if (body.evidence_json !== undefined) {
        try { body.evidence = JSON.parse(body.evidence_json || "[]"); }
        catch { throw new Error("Evidence JSON is invalid"); }
        delete body.evidence_json;
      }
      if (body.only_new !== undefined && typeof body.only_new === "string") {
        body.only_new = ["1", "true", "yes", "y"].includes(String(body.only_new).toLowerCase());
      }
      if (body.target_platforms !== undefined && typeof body.target_platforms === "string") {
        body.target_platforms = body.target_platforms.split(",").map((s) => s.trim()).filter(Boolean);
      }

      let path = ep.path;
      Object.entries(pathVals).forEach(([k, v]) => {
        path = path.replace(`{${k}}`, encodeURIComponent(String(v)));
      });
      const qs = new URLSearchParams(query).toString();
      if (qs) path += (path.includes("?") ? "&" : "?") + qs;
      return { path, body };
    }

    async function api(method, path, body) {
      const headers = { "Content-Type": "application/json" };
      if (state.token) headers.Authorization = `Bearer ${state.token}`;
      const res = await fetch(path, {
        method,
        headers,
        body: ["POST", "PUT", "PATCH"].includes(method) ? JSON.stringify(body ?? {}) : undefined,
      });
      const text = await res.text();
      let data;
      try { data = JSON.parse(text); } catch { data = text; }
      return { ok: res.ok, status: res.status, data };
    }

      meta.className = "meta " + (ok ? "ok" : "bad");
      meta.textContent = title;
      $("output").textContent = typeof data === "string" ? data : JSON.stringify(data, null, 2);
    }

    async function runEndpoint(ep) {
      try {

          showOut(false, "Login first", { detail: "Choose a user and click Login" });

        showOut(result.ok, `${result.status} ${ep.method} ${path}`, result.data);
      } catch (err) {
        showOut(false, "Request error", { detail: String(err.message || err) });
      }
    }

"""Map app errors to HTTP responses (shared by all routers)."""

from __future__ import annotations

from fastapi import HTTPException, status

from utils.errors import AuthError, NotFoundError, PermissionDeniedError, ValidationError


def raise_http_error(exc: Exception) -> None:
    """Convert known app errors to HTTPException. Always raises."""
    if isinstance(exc, AuthError):
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=exc.message,
        ) from exc
    if isinstance(exc, PermissionDeniedError):
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN,
            detail=exc.message,
        ) from exc
    if isinstance(exc, NotFoundError):
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail=exc.message,
        ) from exc
    if isinstance(exc, FileNotFoundError):
        raise HTTPException(
            status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
            detail=str(exc),
        ) from exc
    if isinstance(exc, ValidationError):
        detail = getattr(exc, "message", None) or str(exc)
        if "No trained model" in detail:
            raise HTTPException(
                status_code=status.HTTP_503_SERVICE_UNAVAILABLE,
                detail=detail,
            ) from exc
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail=detail,
        ) from exc
    raise exc

        import json

        flags = json.loads(flags)
    return DraftResponse(
        id=int(row["id"]),
        platform=str(row["platform"]),
        version_number=int(row["version_number"]),
        adapted_title=str(row["adapted_title"]),

        validation_flags=dict(flags),
        created_at=row.get("created_at"),
    )

"""User domain service — employee/user/desk creation with validation."""

from __future__ import annotations

from typing import Any

from config.constants import CAP_MANAGE_USERS, ROLE_ADMIN, ROLE_NEWSROOM_MANAGER
from entities.employee import Employee
from entities.user import User
from modules.authentication.auth_service import admin_reset_password
from modules.authentication.security import hash_password
from repository.user_repository import UserRepository
from services.access import require_capability_session
from utils.errors import NotFoundError, ValidationError
from utils.validators import (
    require_email,
    require_nonempty,
    require_password,
    require_role_code,
    require_username,
)


class UserService:
    def __init__(self, repo: UserRepository | None = None) -> None:
        self.repo = repo or UserRepository()

    def add_user(
        self,
        *,
        username: str,
        password: str,
        first_name: str,
        last_name: str,
        email: str,
        role_code: str = "journalist",
        employee_code: str | None = None,
        session_token: str | None = None,
    ) -> dict[str, Any]:
        actor = require_capability_session(CAP_MANAGE_USERS, session_token)
        username = require_username(username)
        password = require_password(password)
        first_name = require_nonempty(first_name, "First name")
        last_name = require_nonempty(last_name, "Last name")
        email = require_email(email)
        role_code = require_role_code(role_code)
        if actor["role_code"] == ROLE_NEWSROOM_MANAGER and role_code == ROLE_ADMIN:
            raise ValidationError("Newsroom managers cannot create administrator accounts.")

        if self.repo.find_by_username(username):
            raise ValidationError(f"Username '{username}' already exists.")

        role_id = self.repo.role_id_by_code(role_code)
        if role_id is None:
            raise ValidationError(f"Unknown role code: {role_code}")

        code = (employee_code or f"EMP-{username.upper()}").strip()
        emp_id = self.repo.create_employee(
            Employee(
                employee_code=code,
                first_name=first_name,
                last_name=last_name,
                email=email,
                department="Newsroom",
                designation=role_code,
            )
        )
        user_id = self.repo.create_user(
            User(
                employee_id=emp_id,
                role_id=role_id,
                username=username,
                password_hash=hash_password(password),
                account_status="active",
            )
        )
        self.repo.ensure_desk(user_id, role_code)
        return {"user_id": user_id, "username": username, "role_code": role_code}

    def find_user(self, username: str) -> dict[str, Any]:
        username = require_username(username)
        user = self.repo.find_by_username(username)
        if not user:
            raise NotFoundError(f"User '{username}' was not found.")
        return user

    def reset_password(
        self,
        username: str,
        new_password: str,
        *,
        session_token: str | None = None,
    ) -> dict[str, Any]:
        """Password reset for another account (manage_users capability)."""
        require_capability_session(CAP_MANAGE_USERS, session_token)
        return admin_reset_password(username, new_password)

"""Audience Insight Analyst — scores, category, story pitches.

Pitch importance = simple weighted formula (NOT the editor queue urgency model).
Editor queue urgency lives in ml/editor/queue_urgency.py.
"""

from __future__ import annotations

from typing import Any

from database.connection import db_cursor
from entities.content.story_pitch import StoryPitch
from ml.aia import predict_engagement, predict_interest, predict_trend
from repository.content_repository import ContentRepository
from utils.errors import ValidationError


def _pitch_importance(
    trend_score: float,
    audience_interest_score: float,
    coverage_gap_score: float,
) -> str:
    """Combine scores into low / medium / high pitch importance."""
    score = (trend_score * 0.4) + (audience_interest_score * 0.4) + (coverage_gap_score * 0.2)
    if score >= 80:
        return "high"
    if score >= 50:
        return "medium"
    return "low"


class AIAService:
    def __init__(self, repo: ContentRepository | None = None) -> None:
        self.repo = repo or ContentRepository()

    def analyze_engagement(
        self,
        *,
        views: float = 120,
        likes: float = 25,
        shares: float = 8,
        comments: float = 4,
    ) -> dict[str, Any]:
        return predict_engagement(views=views, likes=likes, shares=shares, comments=comments)

    def analyze_interest(
        self,
        *,
        views: float = 120,
        likes: float = 25,
        shares: float = 8,
        comments: float = 4,
    ) -> dict[str, Any]:
        return predict_interest(views=views, likes=likes, shares=shares, comments=comments)

    def analyze_trends(self, title: str = "", summary: str = "") -> dict[str, Any]:
        headline = title or "Audience trend scan across recent newsroom coverage"
        return predict_trend(headline, summary)

    def generate_story_pitch(self, payload: Any) -> dict[str, Any]:
        title = str(payload.title)
        summary = str(payload.summary or "")
        category = str(getattr(payload, "category", "") or "").strip()
        if not category:
            raise ValidationError("category is required (manual entry; no category model).")

        trend_score = float(payload.trend_score or 0)
        interest_score = float(payload.audience_interest_score or 0)
        coverage_gap = float(payload.coverage_gap_score or 0)

        if trend_score <= 0:
            trend_score = float(predict_trend(title, summary)["trend_score"])
        if interest_score <= 0:
            interest_score = float(
                predict_interest(
                    views=float(getattr(payload, "views", 120) or 120),
                    likes=float(getattr(payload, "likes", 25) or 25),
                    shares=float(getattr(payload, "shares", 8) or 8),
                    comments=float(getattr(payload, "comments", 4) or 4),
                )["audience_interest_score"]
            )
        if coverage_gap <= 0:
            coverage_gap = 40.0

        pitch_importance = _pitch_importance(trend_score, interest_score, coverage_gap)

        # DB column is still named "priority"; API uses clearer "pitch_importance"
        pitch = StoryPitch(
            title=title,
            summary=summary,
            priority=pitch_importance,
            source_type="AI_GENERATED",
            trend_score=trend_score,
            audience_interest_score=interest_score,
            coverage_gap_score=coverage_gap,
            status="submitted",
        )
        pitch_id = self.repo.save_pitch(pitch)
        return {
            "story_pitch_id": pitch_id,
            "title": title,
            "summary": summary,
            "pitch_importance": pitch_importance,
            "priority": pitch_importance,
            "importance_source": "formula",
            "source_type": "AI_GENERATED",
            "created_by_desk": "AIA",
            "category": category,
            "trend_score": trend_score,
            "audience_interest_score": interest_score,
            "coverage_gap_score": coverage_gap,
        }

    def get_story_pitches(self) -> dict[str, Any]:
        pitches = self.repo.pitches.list_all()
        return {"pitches": [_annotate_pitch_source(p) for p in pitches]}

    def get_top_story_pitches(self) -> dict[str, Any]:
        pitches = self.repo.pitches.list_all()
        pitches = sorted(
            pitches,
            key=lambda row: (
                float(row.get("trend_score") or 0)
                + float(row.get("audience_interest_score") or 0)
            ),
            reverse=True,
        )
        return {"pitches": [_annotate_pitch_source(p) for p in pitches[:10]]}

    def performance_analytics(self) -> dict[str, Any]:
        """Close the diagram loop: publication/content outcomes back to AIA."""
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT status, COUNT(*) AS count
                FROM publication
                GROUP BY status
                """
            )
            publication_by_status = {
                str(r["status"]): int(r["count"]) for r in cur.fetchall() or []
            }

            cur.execute(
                """
                SELECT status, COUNT(*) AS count
                FROM content
                GROUP BY status
                """
            )
            content_by_status = {str(r["status"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute(
                """
                SELECT status, COUNT(*) AS count,
                       ROUND(AVG(COALESCE(trend_score, 0))::numeric, 2) AS avg_trend,
                       ROUND(AVG(COALESCE(audience_interest_score, 0))::numeric, 2) AS avg_interest
                FROM story_pitch
                GROUP BY status
                """
            )
            pitch_outcomes = [
                {
                    "status": str(r["status"]),
                    "count": int(r["count"]),
                    "avg_trend": float(r["avg_trend"] or 0),
                    "avg_interest": float(r["avg_interest"] or 0),
                }
                for r in cur.fetchall() or []
            ]

            cur.execute(
                """
                SELECT source_type, COUNT(*) AS count
                FROM story_pitch
                GROUP BY source_type
                """
            )
            pitches_by_source = {
                str(r["source_type"] or "UNKNOWN"): int(r["count"]) for r in cur.fetchall() or []
            }

            cur.execute(
                """
                SELECT c.title, p.status AS publication_status, p.published_at
                FROM publication p
                JOIN content c ON c.content_id = p.content_id
                WHERE p.status IN ('published', 'scheduled')
                ORDER BY COALESCE(p.published_at, p.updated_at) DESC NULLS LAST
                LIMIT 10
                """
            )
            recent_published = [dict(r) for r in cur.fetchall() or []]

        published = publication_by_status.get("published", 0)
        scheduled = publication_by_status.get("scheduled", 0)
        approved_content = content_by_status.get("approved", 0)
        insights = []
        if published + scheduled > 0:
            insights.append(
                f"{published} published and {scheduled} scheduled items — reuse winning topics in new pitches."
            )
        if approved_content:
            insights.append(
                f"{approved_content} approved pieces awaiting adaptation/publish — check coverage gaps."
            )
        aia_pitches = pitches_by_source.get("AI_GENERATED", 0)
        manual_pitches = pitches_by_source.get("MANUAL", 0) + pitches_by_source.get("UNKNOWN", 0)
        insights.append(
            f"Pitch mix: {aia_pitches} AIA-generated vs {manual_pitches} manual/other."
        )

        return {
            "publication_by_status": publication_by_status,
            "content_by_status": content_by_status,
            "pitch_outcomes": pitch_outcomes,
            "pitches_by_source": pitches_by_source,
            "recent_published": recent_published,
            "insights": insights,
            "loop": "publication/content outcomes → AIA performance → story pitch workspace",
        }


def _annotate_pitch_source(pitch: dict[str, Any]) -> dict[str, Any]:
    source = str(pitch.get("source_type") or "").strip().upper() or "UNKNOWN"
    if source in {"AI_GENERATED", "AIA", "AI"}:
        created_by = "AIA"
        source_type = "AI_GENERATED"
    elif source in {"MANUAL", "EDITOR", "HUMAN"}:
        created_by = "MANUAL"
        source_type = "MANUAL"
    else:
        created_by = "UNKNOWN"
        source_type = source
    return {
        **pitch,
        "source_type": source_type,
        "created_by_desk": created_by,
    }

from utils.errors import NotFoundError, ValidationError
from utils.validators import (
    require_article_type,
    require_body,
    require_category_name,
    require_content_status,
    require_nonempty,
    require_optional_str,
    require_pitch_status,
    require_positive_id,
    require_priority,
    require_read_time,
    require_severity,
    require_title,
    validate_content_record,
)


class ContentService:
    def __init__(self, repo: ContentRepository | None = None) -> None:
        self.repo = repo or ContentRepository()
        self.categories = CategoryDAO()

    def add_category(self, name: str, description: str = "") -> int:
        return self.categories.create(
            Category(
                category_name=require_category_name(name),
                description=require_optional_str(description, "Description"),
            )
        )

    def add_pitch(self, title: str, summary: str, status: str = "submitted", priority: str = "medium") -> int:
        return self.repo.save_pitch(
            StoryPitch(
                title=require_title(title),
                summary=require_nonempty(summary, "Summary"),
                status=require_pitch_status(status),
                priority=require_priority(priority),
            )
        )

    def add_news(
        self,
        *,
        title: str,
        body: str,
        summary: str = "",
        event_title: str = "",
        event_type: str = "city",
        event_location: str = "",
        severity: str = "low",
        breaking: bool = False,
        status: str = "draft",
        author_id: int | None = None,
        evidence: list[dict[str, Any]] | None = None,
    ) -> dict[str, Any]:
        status = require_content_status(status)
        title = require_title(title)
        body = require_body(body)
        items = list(evidence or [])
        if not items:
            raise ValidationError("At least one evidence item is required when creating news.")
        errors = validate_content_record(
            {"title": title, "text": body, "content_body": body, "status": status}
        )
        if errors:
            raise ValidationError(
                "Invalid news fields: "
                + ", ".join(errors)
                + f". Allowed status: {', '.join(ALLOWED_CONTENT_STATUSES)}."
            )
        content_id = self.repo.content.create(
            Content(
                author_id=author_id,
                title=title,
                summary=require_optional_str(summary, "Summary"),
                raw_body=body,
                content_body=body,
                status=status,
                content_kind="news",
                source_type="api",
            )
        )
        news_id = self.repo.news.create(
            News(
                content_id=content_id,
                event_title=require_optional_str(event_title, "Event title") or title,
                event_type=require_nonempty(event_type, "Event type"),
                event_location=require_optional_str(event_location, "Event location", max_len=255),
                severity=require_severity(severity),
                breaking_news_flag=bool(breaking),
            )
        )
        evidence_ids: list[int] = []
        for item in items:
            evidence_ids.append(
                self.add_evidence(
                    content_id=content_id,
                    title=str(item.get("title") or ""),
                    evidence_type=str(item.get("evidence_type") or "document"),
                    source_url=str(item.get("source_url") or ""),
                    text=str(item.get("text") or ""),
                    uploaded_by=author_id,
                )
            )
        return {"content_id": content_id, "news_id": news_id, "evidence_ids": evidence_ids}

    def add_article(
        self,
        *,
        title: str,
        body: str,
        summary: str = "",
        article_type: str = "feature",
        read_time: int = 3,
        status: str = "draft",
        author_id: int | None = None,
    ) -> dict[str, Any]:
        status = require_content_status(status)
        title = require_title(title)
        body = require_body(body)
        errors = validate_content_record(
            {"title": title, "text": body, "content_body": body, "status": status}
        )
        if errors:
            raise ValidationError("Invalid article fields: " + ", ".join(errors))
        content_id = self.repo.content.create(
            Content(
                author_id=author_id,
                title=title,
                summary=require_optional_str(summary, "Summary"),
                raw_body=body,
                content_body=body,
                status=status,
                content_kind="article",
                source_type="api",
            )
        )
        article_id = self.repo.articles.create(
            Article(
                content_id=content_id,
                article_type=require_article_type(article_type),
                estimated_read_time=require_read_time(read_time),
            )
        )
        return {"content_id": content_id, "article_id": article_id}

    def add_evidence(
        self,
        *,
        content_id: int,
        title: str,
        evidence_type: str = "document",
        source_url: str = "",
        text: str = "",
        uploaded_by: int | None = None,
    ) -> int:
        self.repo.get_content(content_id)
        return self.repo.save_evidence(
            Evidence(
                content_id=content_id,
                title=require_title(title),
                evidence_type=require_nonempty(evidence_type, "Evidence type"),
                source_url=require_optional_str(source_url, "Source URL"),
                body_text=require_optional_str(text, "Evidence text"),
                uploaded_by=uploaded_by,
            )
        )

    def get_content(self, content_id: int) -> dict[str, Any]:
        return self.repo.get_content(require_positive_id(content_id, "Content ID"))

    def get_content_by_input(self, raw_id: str) -> dict[str, Any]:
        return self.get_content(require_positive_id(raw_id, "Content ID"))

    def list_content(self, limit: int = 10, status: str | None = None) -> list[dict[str, Any]]:
        limit = max(1, min(int(limit), 500))
        if status:
            return self.repo.content.list_by_status(require_content_status(status), limit=limit)
        return self.repo.content.list_all(limit=limit)

    def list_news(self, limit: int = 10) -> list[dict[str, Any]]:
        return self.repo.news.list_all(limit=max(1, min(int(limit), 500)))

    def list_raw_ingest(self, limit: int = 15) -> list[dict[str, Any]]:
        return self.repo.raw.list_recent(limit=max(1, min(int(limit), 500)))

    def raw_ingest_counts(self) -> dict[str, int]:
        return self.repo.raw.count_by_status()

    def update_content_status(self, content_id: int, status: str) -> dict[str, Any]:
        content_id = require_positive_id(content_id, "Content ID")
        status = require_content_status(status)
        self.repo.get_content(content_id)
        changed = self.repo.content.update_status(content_id, status)
        if changed != 1:
            raise NotFoundError(f"Content ID {content_id} was not updated.")
        row = self.repo.get_content(content_id)
        return {"content_id": content_id, "status": row["status"], "title": row.get("title")}

"""Newsroom manager — ops overview, bottlenecks, business analytics."""

from __future__ import annotations

from typing import Any

from dao.content_dao import ContentDAO
from dao.publication_dao import PublicationDAO

class ManagerService:
    def __init__(self) -> None:
        self.editor = EditorService()
        self.pitches = StoryPitchDAO()
        self.content = ContentDAO()
        self.publications = PublicationDAO()

    def overview(self) -> dict[str, Any]:
        submitted_pitches = self.pitches.list_by_status("submitted", limit=500)
        submitted_content = self.content.list_by_status("submitted", limit=500)
        approved_content = self.content.list_by_status("approved", limit=500)
        analytics = self.publications.analytics()
        return {
            "submitted_pitches": len(submitted_pitches),
            "submitted_content": len(submitted_content),
            "approved_content": len(approved_content),
            "publication": analytics,
        }

    def list_pitches(self, status: str = "submitted", limit: int = 100) -> dict[str, Any]:
        return self.editor.list_pitches(status=status, limit=limit)

    def list_content(self, status: str = "submitted", limit: int = 100) -> dict[str, Any]:
        return self.editor.list_content(status=status, limit=limit)

    def review_queue(
        self,
        *,
        pitch_status: str = "submitted",
        content_status: str = "submitted",
        limit: int = 100,

            pitch_status=pitch_status,

            limit=limit,

        return self.editor.reject_pitch(story_pitch_id, remarks=remarks)

    def approve_content(self, content_id: int) -> dict[str, Any]:
        return self.editor.approve_content(content_id)

    def reject_content(self, content_id: int, remarks: str = "") -> dict[str, Any]:

    def bottlenecks(self) -> dict[str, Any]:
        """Where work is piling up across the newsroom pipeline."""
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT status AS stage,
                       COUNT(*) AS item_count,
                       ROUND(AVG(EXTRACT(EPOCH FROM (NOW() - updated_at)) / 3600.0)::numeric, 2)
                         AS avg_hours_in_stage
                FROM content
                WHERE status NOT IN ('published', 'draft')
                GROUP BY status
                ORDER BY item_count DESC
                """
            )
            content_stages = [dict(r) for r in cur.fetchall() or []]

            cur.execute(
                """
                SELECT status AS stage,
                       COUNT(*) AS item_count
                FROM story_pitch
                GROUP BY status
                ORDER BY item_count DESC
                """
            )
            pitch_stages = [dict(r) for r in cur.fetchall() or []]

            cur.execute(
                """
                SELECT status AS stage,
                       COUNT(*) AS item_count
                FROM publication
                GROUP BY status
                ORDER BY item_count DESC
                """
            )
            publication_stages = [dict(r) for r in cur.fetchall() or []]

            try:
                cur.execute(
                    """
                    SELECT status AS stage,
                           COUNT(*) AS item_count
                    FROM adaptation_drafts
                    GROUP BY status
                    ORDER BY item_count DESC
                    """
                )
                adaptation_stages = [dict(r) for r in cur.fetchall() or []]
            except Exception:  # noqa: BLE001 — table may be missing in older DBs
                adaptation_stages = []

        hotspots = []
        for row in content_stages:
            if str(row.get("stage")) in {"submitted", "rejected"} and int(row.get("item_count") or 0) > 0:
                hotspots.append(
                    {
                        "area": "content",
                        "stage": row["stage"],
                        "item_count": int(row["item_count"]),
                        "avg_hours_in_stage": float(row.get("avg_hours_in_stage") or 0),
                        "note": "Editorial backlog — prioritize with /manager/queue",
                    }
                )
        for row in pitch_stages:
            if str(row.get("stage")) == "submitted" and int(row.get("item_count") or 0) > 0:
                hotspots.append(
                    {
                        "area": "story_pitch",
                        "stage": row["stage"],
                        "item_count": int(row["item_count"]),
                        "note": "Pitch review backlog",
                    }
                )

        return {
            "content_stages": content_stages,
            "pitch_stages": pitch_stages,
            "publication_stages": publication_stages,
            "adaptation_stages": adaptation_stages,
            "hotspots": hotspots,
        }

    def business_analytics(self) -> dict[str, Any]:
        """Whole-application business snapshot for the newsroom manager."""
        with db_cursor() as cur:
            cur.execute("SELECT status, COUNT(*) AS count FROM content GROUP BY status")
            content_by_status = {str(r["status"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute("SELECT content_kind, COUNT(*) AS count FROM content GROUP BY content_kind")
            content_by_kind = {str(r["content_kind"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute("SELECT status, COUNT(*) AS count FROM story_pitch GROUP BY status")
            pitches_by_status = {str(r["status"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute("SELECT priority, COUNT(*) AS count FROM story_pitch GROUP BY priority")
            pitches_by_priority = {str(r["priority"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute(
                """
                SELECT cleaning_status, COUNT(*) AS count
                FROM raw_ingest
                GROUP BY cleaning_status
                """
            )
            ingest_by_status = {str(r["cleaning_status"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute("SELECT COUNT(*) AS count FROM users WHERE account_status = 'active'")
            active_users = int(cur.fetchone()["count"])

            cur.execute(
                """
                SELECT r.role_code, COUNT(u.user_id) AS count
                FROM role r
                LEFT JOIN users u ON u.role_id = r.role_id AND u.account_status = 'active'
                GROUP BY r.role_code
                ORDER BY r.role_code
                """
            )
            users_by_role = {str(r["role_code"]): int(r["count"]) for r in cur.fetchall() or []}

            cur.execute("SELECT COUNT(*) AS count FROM evidence")
            evidence_count = int(cur.fetchone()["count"])

            cur.execute("SELECT COUNT(*) AS count FROM ingest_file")
            ingest_files = int(cur.fetchone()["count"])

        publication = self.publications.analytics()
        bottlenecks = self.bottlenecks()
        workload = self.workload()
        deadline_risk = self.deadline_risk()
        resources = self.resources()

        return {
            "pipeline": {
                "raw_ingest": ingest_by_status,
                "content": content_by_status,
                "content_kinds": content_by_kind,
                "story_pitches": pitches_by_status,
                "pitch_priority": pitches_by_priority,
                "publication": publication,
            },
            "ops": {
                "active_users": active_users,
                "users_by_role": users_by_role,
                "evidence_items": evidence_count,
                "ingest_files": ingest_files,
            },
            "bottleneck_hotspots": bottlenecks.get("hotspots") or [],
            "workload_summary": {
                "journalist_count": len(workload.get("journalists") or []),
                "total_open_items": workload.get("total_open_items"),
            },
            "deadline_risk_summary": {
                "at_risk_count": deadline_risk.get("at_risk_count"),
                "thresholds_hours": deadline_risk.get("thresholds_hours"),
            },
            "resource_summary": resources.get("utilization"),
            "totals": {
                "content": sum(content_by_status.values()),
                "story_pitches": sum(pitches_by_status.values()),
                "publications": sum((publication.get("publications_by_status") or {}).values()),
            },
        }

    def workload(self) -> dict[str, Any]:
        """Per-journalist open workload + availability."""
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT j.journalist_id, j.user_id, u.username,
                       COALESCE(NULLIF(TRIM(e.first_name || ' ' || e.last_name), ''), u.username)
                         AS full_name,
                       j.primary_specialization, j.secondary_specialization,
                       j.availability_status,
                       COUNT(c.content_id) FILTER (
                         WHERE c.status IN ('draft', 'submitted', 'rework')
                       ) AS open_items,
                       COUNT(c.content_id) FILTER (WHERE c.status = 'rework') AS rework_items,
                       COUNT(c.content_id) FILTER (WHERE c.status = 'submitted') AS submitted_items
                FROM journalist j
                JOIN users u ON u.user_id = j.user_id
                LEFT JOIN employee e ON e.employee_id = u.employee_id
                LEFT JOIN content c ON c.author_id = j.journalist_id
                WHERE j.status = 'active'
                GROUP BY j.journalist_id, j.user_id, u.username, e.first_name, e.last_name,
                         j.primary_specialization, j.secondary_specialization, j.availability_status
                ORDER BY open_items DESC, j.journalist_id
                """
            )
            journalists = [dict(r) for r in cur.fetchall() or []]
        total_open = sum(int(j.get("open_items") or 0) for j in journalists)
        return {
            "journalists": journalists,
            "total_open_items": total_open,
            "method": "open content in draft/submitted/rework by author_id",
        }

    def deadline_risk(self, *, submitted_hours: float = 48.0, rework_hours: float = 24.0) -> dict[str, Any]:
        """Items stuck too long in stage (proxy for deadline risk)."""
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT content_id, title, status, author_id,
                       ROUND(EXTRACT(EPOCH FROM (NOW() - COALESCE(submitted_at, updated_at))) / 3600.0)
                         AS hours_in_stage
                FROM content
                WHERE status IN ('submitted', 'rework')
                  AND EXTRACT(EPOCH FROM (NOW() - COALESCE(submitted_at, updated_at))) / 3600.0
                      >= CASE WHEN status = 'rework' THEN %s ELSE %s END
                ORDER BY hours_in_stage DESC
                LIMIT 200
                """,
                (rework_hours, submitted_hours),
            )
            content_risk = [dict(r) for r in cur.fetchall() or []]

            cur.execute(
                """
                SELECT story_pitch_id, title, status, priority, source_type,
                       ROUND(EXTRACT(EPOCH FROM (NOW() - COALESCE(submitted_at, created_at))) / 3600.0)
                         AS hours_in_stage
                FROM story_pitch
                WHERE status = 'submitted'
                  AND EXTRACT(EPOCH FROM (NOW() - COALESCE(submitted_at, created_at))) / 3600.0
                      >= %s
                ORDER BY hours_in_stage DESC
                LIMIT 200
                """,
                (submitted_hours,),
            )
            pitch_risk = [dict(r) for r in cur.fetchall() or []]

        return {
            "thresholds_hours": {
                "submitted": submitted_hours,
                "rework": rework_hours,
            },
            "content_at_risk": content_risk,
            "pitches_at_risk": pitch_risk,
            "at_risk_count": len(content_risk) + len(pitch_risk),
            "method": "age in stage vs thresholds (no due_at column)",
        }

    def resources(self) -> dict[str, Any]:
        """Resource utilization across desks + journalist load distribution."""
        workload = self.workload()
        journalists = workload.get("journalists") or []
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT r.role_code, r.role_name, COUNT(u.user_id) AS active_users
                FROM role r
                LEFT JOIN users u ON u.role_id = r.role_id AND u.account_status = 'active'
                GROUP BY r.role_code, r.role_name
                ORDER BY r.role_code
                """
            )
            desks = [dict(r) for r in cur.fetchall() or []]

        loads = [int(j.get("open_items") or 0) for j in journalists]
        available = sum(
            1 for j in journalists if str(j.get("availability_status") or "") == "available"
        )
        avg_load = round(sum(loads) / len(loads), 2) if loads else 0.0
        overloaded = [j for j in journalists if int(j.get("open_items") or 0) >= max(3, avg_load + 2)]
        underutilized = [
            j
            for j in journalists
            if str(j.get("availability_status") or "") == "available"
            and int(j.get("open_items") or 0) == 0
        ]

        return {
            "desks": desks,
            "utilization": {
                "active_journalists": len(journalists),
                "available_journalists": available,
                "avg_open_load": avg_load,
                "overloaded_count": len(overloaded),
                "underutilized_count": len(underutilized),
            },
            "overloaded": overloaded[:20],
            "underutilized": underutilized[:20],
            "method": "role headcount + journalist open-load distribution",
        }

"""Journalist tools façade — capability check, then call a service."""

from __future__ import annotations

from typing import Any, Mapping, Sequence

from config.constants import CAP_JOURNALIST_TOOLS
from dao.journalist_dao import JournalistDAO
from services.access import require_capability_session
from services.content import ContentService
from services.journalist import (
    assess_credibility,
    extract_metadata,
    find_duplicates,
    search_evidence,
    search_patents,
    search_related_news,
    validate_article,
)


class JournalistModule:
    def __init__(
        self,
        session_token: str | None = None,
        content: ContentService | None = None,
    ) -> None:
        self.session_token = session_token
        self.content = content or ContentService()

    def _user(self) -> dict[str, Any]:
        return require_capability_session(CAP_JOURNALIST_TOOLS, self.session_token)

    def _author_id(self, user: dict[str, Any]) -> int | None:
        """content.author_id FK → journalist.journalist_id (not users.user_id)."""
        profiles = user.get("profiles") or {}
        journalist_id = profiles.get("journalist_id")
        if journalist_id:
            return int(journalist_id)
        return JournalistDAO().ensure_for_user(int(user["user_id"]))

    def research(
        self,
        query: str,
        documents: Sequence[Mapping[str, Any]] | None = None,
        *,
        limit: int = 10,
    ) -> dict[str, Any]:
        user = self._user()
        hits = search_related_news(query, documents, limit=limit)
        return {"actor": user["username"], "results": hits}

    def evidence_search(
        self,
        query: str,
        documents: Sequence[Mapping[str, Any]] | None = None,
        *,
        limit: int = 10,
    ) -> dict[str, Any]:
        user = self._user()
        hits = search_evidence(query, documents, limit=limit)
        return {"actor": user["username"], "results": hits}

    def metadata(self, text: str) -> dict[str, Any]:
        user = self._user()
        return {"actor": user["username"], "entities": extract_metadata(text)}

    def duplicates(
        self,
        documents: Sequence[Mapping[str, Any]] | None = None,
        *,
        threshold: float = 0.5,
        limit: int = 20,
    ) -> dict[str, Any]:
        user = self._user()
        pairs = find_duplicates(documents, threshold=threshold, limit=limit)
        return {"actor": user["username"], "pairs": pairs}

    def validate(self, record: Mapping[str, Any] | None = None) -> dict[str, Any]:
        user = self._user()
        return {"actor": user["username"], **validate_article(record)}

    def credibility(
        self,
        text: str,
        evidence_docs: Sequence[Mapping[str, Any]] | None = None,
        *,
        limit: int = 5,
    ) -> dict[str, Any]:
        user = self._user()
        report = assess_credibility(text, evidence_docs, limit=limit)
        return {"actor": user["username"], **report}

    def patents(
        self,
        query: str,
        documents: Sequence[Mapping[str, Any]] | None = None,
        *,
        limit: int = 10,
    ) -> dict[str, Any]:
        user = self._user()
        hits = search_patents(query, documents, limit=limit)
        return {"actor": user["username"], "results": hits}

    # ── Desk content (parity with console journalist menu) ─────────────────

    def add_news(self, **kwargs: Any) -> dict[str, Any]:
        user = self._user()
        kwargs.setdefault("author_id", self._author_id(user))
        created = self.content.add_news(**kwargs)
        return {"actor": user["username"], **created}

    def add_article(self, **kwargs: Any) -> dict[str, Any]:
        user = self._user()
        kwargs.setdefault("author_id", self._author_id(user))
        created = self.content.add_article(**kwargs)
        return {"actor": user["username"], **created}

    def list_content(self, limit: int = 20, status: str | None = None) -> dict[str, Any]:
        user = self._user()
        return {
            "actor": user["username"],
            "items": self.content.list_content(limit=limit, status=status),
        }

    def related_content(
        self,
        title: str = "",
        text: str = "",
        *,
        limit: int = 8,
    ) -> dict[str, Any]:
        from ml.knowledge import score_content_relevancy

        user = self._user()
        result = score_content_relevancy(title, text, limit=limit)
        return {
            "actor": user["username"],
            "matches": result.get("matches") or [],
            "relevancy": result.get("relevancy") or "none",
            "best_score": float(result.get("best_score") or 0.0),
            "knowledge_base_size": int(result.get("knowledge_base_size") or 0),

"""Editor queue urgency model.

Tells the editor how urgent a story is for the review queue:
  urgent > high > medium > low

This is NOT the same as AIA "pitch importance" (that one is a simple formula).
"""

from __future__ import annotations

import re
from functools import lru_cache
from pathlib import Path
from typing import Any

import numpy as np
from scipy.sparse import csr_matrix, hstack

from utils.logger import get_logger

log = get_logger("ml.editor.queue_urgency")

ROOT = Path(__file__).resolve().parents[2]
MODEL_PATHS = (
    ROOT / "artifacts" / "models" / "editor_queue_urgency.joblib",
)

# Model training labels → simple desk levels used in the app
LABEL_TO_URGENCY = {
    "very low": "low",
    "low": "low",
    "medium": "medium",
    "high": "high",
    "very high": "urgent",
}


@lru_cache(maxsize=1)
def _load_model() -> dict[str, Any] | None:
    path = next((p for p in MODEL_PATHS if p.exists()), None)
    if path is None:
        log.warning("editor_queue_urgency.joblib not found")
        return None
    try:
        import joblib

        bundle = joblib.load(path)
        if not {"model", "tfidf", "labels", "urg_regex"}.issubset(bundle):
            log.warning("Urgency model file is missing required keys: %s", path)
            return None
        log.info("Loaded editor queue urgency model from %s", path)
        return bundle
    except Exception as exc:  # noqa: BLE001
        log.warning("Could not load urgency model: %s", exc)
        return None


def model_available() -> bool:
    return _load_model() is not None


def clear_cache() -> None:
    _load_model.cache_clear()


def _extra_features(texts: list[str], urg_regex: str) -> np.ndarray:
    """Simple extra signals: length, urgent words, digits, ALLCAPS runs."""
    rows = []
    urg = re.compile(urg_regex, re.I)
    digit = re.compile(r"\d")
    caps = re.compile(r"[A-Z]{4,}")
    for text in texts:
        raw = text or ""
        rows.append(
            [
                len(raw) / 500.0,
                len(urg.findall(raw)),
                1 if digit.search(raw) else 0,
                len(caps.findall(raw)),
            ]
        )
    return np.asarray(rows, dtype=float)


def predict_editor_queue_urgency(headline: str, body: str = "") -> dict[str, Any]:
    """Return how urgent this story is for the editor queue."""
    bundle = _load_model()
    text = f"{headline or ''}. {body or ''}".strip()

    if bundle is None:
        return {
            "urgency": "medium",
            "model_label": None,
            "confidence": None,
            "probabilities": {},
            "source": "fallback",
        }

    labels: list[str] = list(bundle["labels"])
    features = hstack(
        [
            bundle["tfidf"].transform([text]),
            csr_matrix(_extra_features([text], str(bundle["urg_regex"]))),
        ]
    ).tocsr()

    probs = bundle["model"].predict_proba(features)[0]
    best = int(np.argmax(probs))
    model_label = labels[best]

    return {
        "urgency": LABEL_TO_URGENCY.get(model_label, "medium"),
        "model_label": model_label,
        "confidence": round(float(probs[best]), 4),
        "probabilities": {label: round(float(p), 4) for label, p in zip(labels, probs)},
        "source": "model",
    }

"""AIA audience scores — formulas only (no category model).

1) Engagement score  — formula from views/likes/shares/comments
2) Audience interest — formula from how intense the interactions are
3) Trend score       — heuristic from headline/body cues
"""

from __future__ import annotations

import re
from typing import Any

TREND_WORDS = re.compile(
    r"\b(breaking|surge|record|viral|trending|boom|crisis|election|finals|"
    r"championship|outbreak|ban|protest|strike|launch)\b",
    re.I,
)


def predict_engagement(
    *,
    views: float = 120,
    likes: float = 25,
    shares: float = 8,
    comments: float = 4,
) -> dict[str, Any]:
    """Simple engagement score from interaction counts."""
    raw = likes * 2.0 + shares * 3.0 + comments * 4.0 + views * 0.05
    score = round(min(100.0, max(0.0, raw / 10.0)), 2)
    return {
        "engagement_score": score,
        "source": "heuristic",
        "raw": round(raw, 4),
        "features_used": ["views", "likes", "shares", "comments"],
    }


def predict_interest(
    *,
    views: float = 120,
    likes: float = 25,
    shares: float = 8,
    comments: float = 4,
    engagement_raw: float | None = None,
) -> dict[str, Any]:
    """Simple audience-interest score (interactions per view)."""
    if engagement_raw is None:
        engagement_raw = float(
            predict_engagement(views=views, likes=likes, shares=shares, comments=comments)["raw"]
        )

    intensity = (likes + shares * 2.0 + comments * 2.0) / max(float(views), 1.0)
    score = round(min(100.0, max(0.0, intensity * 100.0)), 2)

    if score >= 70:
        segment = 2  # high interest
    elif score >= 35:
        segment = 1  # medium
    else:
        segment = 0  # low

    return {
        "audience_interest_score": score,
        "source": "heuristic",
        "segment": segment,
        "distance": None,
        "engagement_raw": round(float(engagement_raw), 4),
    }


def predict_trend(headline: str = "", body: str = "") -> dict[str, Any]:
    """Trend score from simple text cues (no category model)."""
    text = f"{headline or ''}. {body or ''}".strip()
    hits = len(TREND_WORDS.findall(text))
    length_boost = min(15.0, len(text.split()) / 20.0)
    score = round(min(100.0, max(20.0, 45.0 + hits * 12.0 + length_boost)), 2)
    return {
        "trend_score": score,
        "source": "heuristic",
        "category": None,
        "category_confidence": None,
    }

"""Select sentences for a target platform using the unified CAE model."""

from __future__ import annotations

from config.constants import PLATFORM_SENTENCE_LIMITS
from ml.adaptation.model_loader import load_model
from ml.adaptation.sentences import split_sentences
from utils.errors import ValidationError

DEPENDENT_STARTERS = ("however", "but", "therefore", "this", "they", "it", "he", "she", "although")


def predict_sentences(
    title: str,
    content: str,
    platform: str,
    threshold_delta: float = 0.0,
    sentence_delta: int = 0,
    prioritize_quotes: bool = False,
    prioritize_numbers: bool = False,
) -> tuple[list[tuple[int, str, float]], dict, float]:
    sentences = split_sentences(content)
    if not sentences:
        raise ValidationError("No valid sentences found in original content.")

    try:
        model, meta = load_model(platform)
    except FileNotFoundError as exc:
        raise ValidationError(str(exc)) from exc

    platform_key = str(platform).upper()
    records = []
    for i, sentence in enumerate(sentences):
        prev = sentences[i - 1] if i > 0 else ""
        nxt = sentences[i + 1] if i + 1 < len(sentences) else ""
        records.append(
            {
                "headline": title,
                "sentence": sentence,
                "context": f"PREVIOUS {prev} CURRENT {sentence} NEXT {nxt}",
                "position": i,
                "total_sentences": len(sentences),
                "platform": platform_key,
            }
        )
    probabilities = [float(p) for p in model.predict_proba(records)[:, 1]]

    platform_thresholds = meta.get("platform_thresholds") or {}
    base = float(
        platform_thresholds.get(platform_key, meta.get("threshold", 0.5))
    )
    threshold = max(0.05, min(0.95, base + threshold_delta))
    model_version = str(meta.get("version", "1.0.0"))

    scored: list[tuple[int, str, float]] = []
    for i, (sentence, prob) in enumerate(zip(sentences, probabilities)):
        boost = 0.0
        if prioritize_quotes and ('"' in sentence or "“" in sentence):
            boost += 0.15
        if prioritize_numbers and any(ch.isdigit() for ch in sentence):
            boost += 0.15
        scored.append((i, sentence, min(1.0, float(prob) + boost)))

    max_count = max(1, int(PLATFORM_SENTENCE_LIMITS.get(platform, 5)) + int(sentence_delta))
    selected = [row for row in scored if row[2] >= threshold]
    if not selected:
        selected = sorted(scored, key=lambda row: row[2], reverse=True)[:1]
    selected = sorted(selected, key=lambda row: row[2], reverse=True)[:max_count]

    indexes = {row[0] for row in selected}
    for i in list(indexes):
        if i > 0 and sentences[i].lower().startswith(DEPENDENT_STARTERS):
            indexes.add(i - 1)

    ordered = [(i, sentences[i], float(probabilities[i])) for i in sorted(indexes)]
    meta = {**meta, "version": model_version, "platform": platform_key}
    return ordered, meta, threshold

def grouped_split(sentence_df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame, pd.DataFrame]:
    first_split = GroupShuffleSplit(n_splits=1, test_size=0.30, random_state=42)
    train_indexes, remaining_indexes = next(
        first_split.split(sentence_df, groups=sentence_df["article_id"])
    )
    train_df = sentence_df.iloc[train_indexes].reset_index(drop=True)
    remaining_df = sentence_df.iloc[remaining_indexes].reset_index(drop=True)
    second_split = GroupShuffleSplit(n_splits=1, test_size=0.50, random_state=42)
    validation_indexes, test_indexes = next(
        second_split.split(remaining_df, groups=remaining_df["article_id"])
    )
    validation_df = remaining_df.iloc[validation_indexes].reset_index(drop=True)
    test_df = remaining_df.iloc[test_indexes].reset_index(drop=True)
    return train_df, validation_df, test_df


def evaluate(pipeline: Pipeline, frame: pd.DataFrame, threshold: float) -> dict:
    features = frame.to_dict("records")
    actual = frame["label"].to_numpy()
    probabilities = pipeline.predict_proba(features)[:, 1]
    predicted = (probabilities >= threshold).astype(int)
    precision, recall, f1, _ = precision_recall_fscore_support(
        actual, predicted, average="binary", zero_division=0
    )
    ap = (
        float(average_precision_score(actual, probabilities))
        if len(set(actual)) >= 2
        else 0.0
    )
    return {
        "accuracy": float(accuracy_score(actual, predicted)),
        "precision": float(precision),
        "recall": float(recall),
        "f1": float(f1),
        "average_precision": ap,
        "confusion_matrix": confusion_matrix(actual, predicted, labels=[0, 1]).tolist(),
        "records": int(len(frame)),
    }


def _print_matrix(label: str, split: str, metrics: dict) -> None:

    print(f"Records: {metrics['records']}")
    print("Confusion matrix [[TN, FP], [FN, TP]]:")
    print(f"  {cm[0]}")
    print(f"  {cm[1]}")


def validate_source_dataset(dataframe: pd.DataFrame) -> pd.DataFrame:
    required = {
        "article_id",
        "original_title",
        "original_content",
        "target_platform",
        "correct_adapted_article",
    }
    missing = required - set(dataframe.columns)
    if missing:
        raise ValueError(f"Missing columns: {sorted(missing)}")
    cleaned = dataframe.copy()
    for col in required:
        cleaned[col] = cleaned[col].fillna("").astype(str).str.strip()
    empty = cleaned[
        (cleaned["article_id"] == "")
        | (cleaned["original_content"] == "")
        | (cleaned["correct_adapted_article"] == "")
    ]
    if not empty.empty:
        raise ValueError(f"{len(empty)} rows missing required text fields.")
    supported = set(ADAPTATION_PLATFORMS)
    unsupported = set(cleaned["target_platform"].unique()) - supported
    if unsupported:
        raise ValueError(f"Unsupported platforms: {sorted(unsupported)}")
    print()
    print("Source dataset validation")
    print("-" * 70)
    print(f"Dataset rows: {len(cleaned)}")
    print(f"Unique articles: {cleaned['article_id'].nunique()}")
    print(cleaned["target_platform"].value_counts().to_string())
    return cleaned


def train_unified_model(

    print("=" * 70)

    sentence_rows: list[dict] = []
    for row in source_df.to_dict("records"):
        sentence_rows.extend(build_sentence_records(row))
    sentence_df = pd.DataFrame(sentence_rows)
    if sentence_df.empty:
        raise ValueError("No sentence records generated for CAE training.")

    label_counts = sentence_df["label"].value_counts().sort_index().to_dict()
    article_count = int(sentence_df["article_id"].nunique())

    print(f"Unique articles: {article_count}")
    print(f"Sentence records: {len(sentence_df)}")

    print(f"Platforms: {sorted(sentence_df['platform'].unique())}")
    if sentence_df["label"].nunique() < 2:

    train_df, validation_df, test_df = grouped_split(sentence_df)
    for name, frame in (
        ("training", train_df),
        ("validation", validation_df),
        ("testing", test_df),
    ):
        counts = frame["label"].value_counts().sort_index().to_dict()

    pipeline = Pipeline(

    metadata_path = platform_folder / "metadata.json"
    metrics_path = platform_folder / "metrics.json"

        "feature_extractor": "Word TF-IDF + Character TF-IDF + Manual Features + Platform",

        "test_records": int(len(test_df)),

"""Sentence features for the unified CAE retention model (TF-IDF + manual + platform)."""

from __future__ import annotations

import re

import numpy as np
from scipy.sparse import csr_matrix, hstack
from sklearn.base import BaseEstimator, TransformerMixin
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.preprocessing import MaxAbsScaler

from config.constants import ADAPTATION_PLATFORMS

NEGATION = re.compile(r"\b(no|not|never|neither|without)\b", re.I)
UNCERTAINTY = re.compile(
    r"\b(may|might|could|expected|likely|possible|estimated)\b", re.I
)
ATTRIBUTION = re.compile(r"\b(said|stated|reported|announced|according)\b", re.I)
CONDITION = re.compile(
    r"\b(if|unless|subject to|provided|depending|after approval)\b", re.I
)
CONTRAST = re.compile(r"\b(but|however|although|despite|while)\b", re.I)
NUMBER = re.compile(r"\b\d[\d,]*(?:\.\d+)?\b")
PERCENT = re.compile(r"(?:\d+(?:\.\d+)?\s*%|\d+(?:\.\d+)?\s+percent)", re.I)
CURRENCY = re.compile(
    r"(?:₹|\$|£|€)\s*\d|\b\d+(?:\.\d+)?\s*(?:crore|lakh|million|billion)\b", re.I
)
DATE = re.compile(
    r"\b(?:Jan(?:uary)?|Feb(?:ruary)?|Mar(?:ch)?|Apr(?:il)?|May|Jun(?:e)?|"
    r"Jul(?:y)?|Aug(?:ust)?|Sep(?:tember)?|Oct(?:ober)?|Nov(?:ember)?|"
    r"Dec(?:ember)?)\s+\d{1,2}\b",
    re.I,
)


def numeric_features(records: list[dict]) -> np.ndarray:
    rows = []
    for r in records:
        s = r["sentence"]
        i = r["position"]
        n = max(r["total_sentences"] - 1, 1)
        platform = str(r.get("platform") or "").upper()
        one_hot = [1.0 if platform == code else 0.0 for code in ADAPTATION_PLATFORMS]
        rows.append(
            [
                i / n,
                len(s.split()),
                int(i == 0),
                int(i == 1),
                bool(NUMBER.search(s)),
                bool(PERCENT.search(s)),
                bool(CURRENCY.search(s)),
                bool(DATE.search(s)),
                int('"' in s or "“" in s),
                bool(NEGATION.search(s)),
                bool(UNCERTAINTY.search(s)),
                bool(ATTRIBUTION.search(s)),
                bool(CONDITION.search(s)),
                bool(CONTRAST.search(s)),
                *one_hot,
            ]
        )
    return np.asarray(rows, dtype=float)


class SentenceFeatureExtractor(BaseEstimator, TransformerMixin):
    def __init__(self, max_word_features: int = 30000, max_char_features: int = 20000):
        self.max_word_features = max_word_features
        self.max_char_features = max_char_features
        self.word = TfidfVectorizer(
            lowercase=True,
            stop_words="english",
            ngram_range=(1, 3),
            min_df=1,
            max_df=0.98,
            sublinear_tf=True,
            max_features=max_word_features,
        )
        self.char = TfidfVectorizer(
            analyzer="char_wb",
            ngram_range=(3, 5),
            min_df=1,
            sublinear_tf=True,
            max_features=max_char_features,
        )
        self.scaler = MaxAbsScaler()

    def fit(self, X, y=None):
        contexts = [r["context"] for r in X]
        self.word.fit(contexts)
        self.char.fit(contexts)
        self.scaler.fit(numeric_features(X))
        return self

    def transform(self, X):
        contexts = [r["context"] for r in X]
        a = self.word.transform(contexts)
        b = self.char.transform(contexts)
        c = csr_matrix(self.scaler.transform(numeric_features(X)))
        return hstack([a, b, c], format="csr")

"""Sentence splitting for adaptation."""

from __future__ import annotations

import re

ABBREVIATION_PATTERN = re.compile(r"\b(?:Mr|Mrs|Ms|Dr|Prof|Inc|Ltd|No)\.$", re.I)
_WS = re.compile(r"\s+")


def normalize_space(text: str) -> str:
    return _WS.sub(" ", text or "").strip()


def split_sentences(text: str) -> list[str]:
    text = normalize_space(text)
    if not text:
        return []
    pieces = re.split(r'(?<=[.!?])\s+(?=[A-Z“"])', text)
    result: list[str] = []
    carry = ""
    for piece in pieces:
        piece = f"{carry} {piece}".strip() if carry else piece.strip()
        if ABBREVIATION_PATTERN.search(piece):
            carry = piece
            continue
        carry = ""
        if len(piece.split()) >= 3:
            result.append(piece)
    if carry:
        result.append(carry)
    return result


def context_window(sentences: list[str], position: int) -> str:
    """PREVIOUS / CURRENT / NEXT string used by CAE feature extractor."""
    prev = sentences[position - 1] if position > 0 else ""
    current = sentences[position] if 0 <= position < len(sentences) else ""
    nxt = sentences[position + 1] if position + 1 < len(sentences) else ""
    return f"PREVIOUS {prev} CURRENT {current} NEXT {nxt}"

"""Build sentence retain/reject labels from paired original ↔ adapted articles."""

from __future__ import annotations

from typing import Any

import numpy as np
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

from ml.adaptation.sentences import context_window, split_sentences

STRONG_POSITIVE_THRESHOLD = 0.50
NEGATIVE_THRESHOLD = 0.20
MAX_POSITIVE_SENTENCES = {
    "SOCIAL_MEDIA": 3,
    "EMAIL_NEWSLETTER": 6,
    "PRINT_NEWSPAPER": 9,
    "WEBSITE": 11,
}
MIN_POSITIVE_SENTENCES = {
    "SOCIAL_MEDIA": 1,
    "EMAIL_NEWSLETTER": 2,
    "PRINT_NEWSPAPER": 3,
    "WEBSITE": 4,
}


def _create_similarity_matrix(
    source_sentences: list[str],
    target_sentences: list[str],
) -> np.ndarray:
    corpus = source_sentences + target_sentences
    word_vectorizer = TfidfVectorizer(
        lowercase=True,
        stop_words="english",
        ngram_range=(1, 2),
        sublinear_tf=True,
    )
    character_vectorizer = TfidfVectorizer(
        lowercase=True,
        analyzer="char_wb",
        ngram_range=(3, 5),
        sublinear_tf=True,
    )
    word_matrix = word_vectorizer.fit_transform(corpus)
    character_matrix = character_vectorizer.fit_transform(corpus)
    source_count = len(source_sentences)
    word_similarity = cosine_similarity(word_matrix[:source_count], word_matrix[source_count:])
    character_similarity = cosine_similarity(
        character_matrix[:source_count], character_matrix[source_count:]
    )
    return 0.70 * word_similarity + 0.30 * character_similarity


def _select_positive_indexes(similarity_matrix: np.ndarray, platform: str) -> set[int]:
    source_count, target_count = similarity_matrix.shape
    if source_count == 0 or target_count == 0:
        return set()

    maximum_scores = similarity_matrix.max(axis=1)
    selected_indexes: set[int] = set()
    for target_index in range(target_count):
        selected_indexes.add(int(np.argmax(similarity_matrix[:, target_index])))
    for source_index, score in enumerate(maximum_scores):
        if score >= STRONG_POSITIVE_THRESHOLD:
            selected_indexes.add(source_index)

    minimum_count = min(MIN_POSITIVE_SENTENCES.get(platform, 1), source_count)
    maximum_count = min(MAX_POSITIVE_SENTENCES.get(platform, max(1, target_count)), source_count)
    ranked_indexes = np.argsort(maximum_scores)[::-1].tolist()
    for source_index in ranked_indexes:
        if len(selected_indexes) >= minimum_count:
            break
        selected_indexes.add(int(source_index))
    if len(selected_indexes) > maximum_count:
        selected_indexes = set(
            sorted(selected_indexes, key=lambda i: maximum_scores[i], reverse=True)[
                :maximum_count
            ]
        )
    return selected_indexes


def build_sentence_records(row: dict[str, Any]) -> list[dict[str, Any]]:
    source_sentences = split_sentences(str(row["original_content"]))
    target_sentences = split_sentences(str(row["correct_adapted_article"]))
    if not source_sentences or not target_sentences:
        return []

    similarity_matrix = _create_similarity_matrix(source_sentences, target_sentences)
    maximum_scores = similarity_matrix.max(axis=1)
    positive_indexes = _select_positive_indexes(
        similarity_matrix, str(row["target_platform"])
    )

    records: list[dict[str, Any]] = []
    for position, sentence in enumerate(source_sentences):
        records.append(
            {
                "article_id": str(row["article_id"]),
                "platform": str(row["target_platform"]),
                "headline": str(row["original_title"]),
                "sentence": sentence,
                "context": context_window(source_sentences, position),
                "position": position,
                "total_sentences": len(source_sentences),
                "label": 1 if position in positive_indexes else 0,
                "label_similarity": float(maximum_scores[position]),
            }
        )
    return records

def train_urgency_from_db(*, fast: bool = False) -> dict:
    """Train urgency using rows from DB; print classification report + confusion matrix."""
    import re

    import joblib
    import numpy as np
    from scipy.sparse import csr_matrix, hstack
    from sklearn.feature_extraction.text import TfidfVectorizer
    from sklearn.metrics import (
        accuracy_score,
        classification_report,
        confusion_matrix,
        f1_score,
    )
    from sklearn.model_selection import RandomizedSearchCV, StratifiedKFold, train_test_split
    from xgboost import XGBClassifier

    LABELS = ["very low", "low", "medium", "high", "very high"]
    URG_WORDS = (
        r"urgent|breaking|critical|immediate|emergency|severe|ban|sanction|"
        r"recall|bankrupt|attack|fire|halt|shutdown|breach"
    )
    SEED = 42

    df = _load_urgency_from_db()
    if df.empty:
        push_urgency()
        df = _load_urgency_from_db()
    print(f"Training urgency from DB ({len(df)} rows)...")

    texts = (df["headline"].fillna("") + ". " + df["body"].fillna("")).to_numpy(dtype=object)
    y = df["priority"].map({label: i for i, label in enumerate(LABELS)}).to_numpy(dtype=int)
    texts_train, texts_test, y_train, y_test = train_test_split(
        texts, y, test_size=0.2, stratify=y, random_state=SEED
    )

    def numeric_features(series_texts):
        s = pd.Series(list(series_texts))
        return np.c_[
            s.str.len().values / 500.0,
            s.str.count(URG_WORDS, flags=re.I).values,
            s.str.contains(r"\d").astype(int).values,
            s.str.count(r"[A-Z]{4,}").values,
        ]

    def featurize(tfidf, series_texts, *, fit=False):
        matrix = tfidf.fit_transform(series_texts) if fit else tfidf.transform(series_texts)
        return hstack([matrix, csr_matrix(numeric_features(series_texts))]).tocsr()

    tfidf = TfidfVectorizer(ngram_range=(1, 2), max_features=5000, min_df=2, sublinear_tf=True)
    X_train = featurize(tfidf, texts_train, fit=True)
    X_test = featurize(tfidf, texts_test)

    base = XGBClassifier(
        objective="multi:softprob",
        num_class=5,
        tree_method="hist",
        eval_metric="mlogloss",
        random_state=SEED,
        n_jobs=-1,
    )
    if fast:
        model = base.set_params(n_estimators=200, max_depth=5, learning_rate=0.1)
        model.fit(X_train, y_train)
        best_params = {"n_estimators": 200, "max_depth": 5, "learning_rate": 0.1}
        print("Best params:", best_params)
        print("CV macro F1: (skipped --fast)")
    else:
        search = RandomizedSearchCV(
            base,
            {
                "n_estimators": [200, 400],
                "max_depth": [3, 5, 7],
                "learning_rate": [0.05, 0.1, 0.2],
                "subsample": [0.8, 1.0],
                "colsample_bytree": [0.5, 0.8],
            },
            n_iter=8,
            scoring="f1_macro",
            cv=StratifiedKFold(3, shuffle=True, random_state=SEED),
            random_state=SEED,
            n_jobs=1,
        )
        search.fit(X_train, y_train)
        model = search.best_estimator_
        print("Best params:", search.best_params_)
        print("CV macro F1:", round(float(search.best_score_), 4))

    pred = model.predict(X_test)
    print("Accuracy:", round(float(accuracy_score(y_test, pred)), 4))
    print("Macro F1:", round(float(f1_score(y_test, pred, average="macro")), 4))
    print("Within 1 level:", round(float(np.mean(np.abs(pred - y_test) <= 1)), 4))
    print(classification_report(y_test, pred, target_names=LABELS, digits=3))
    print(pd.DataFrame(confusion_matrix(y_test, pred), index=LABELS, columns=LABELS))

    art = ROOT / "artifacts" / "models" / "editor_queue_urgency.joblib"
    art.parent.mkdir(parents=True, exist_ok=True)
    payload = {
        "model": model,
        "model_name": "XGBoost",
        "tfidf": tfidf,
        "labels": LABELS,
        "urg_regex": URG_WORDS,
    }
    joblib.dump(payload, art)
    print("Saved", art)
    return {
        "accuracy": float(accuracy_score(y_test, pred)),
        "macro_f1": float(f1_score(y_test, pred, average="macro")),
        "n_train": int(len(y_train)),
        "n_test": int(len(y_test)),
    }

from __future__ import annotations

from pathlib import Path
from typing import Any

import pandas as pd

from ml.common import rank
from utils.logger import get_logger

ROOT = Path(__file__).resolve().parents[2]

    ROOT / "data" / "news" / "current_news.csv",
    ROOT / "data" / "historical" / "historical_headlines.csv",
)


def _docs_from_csv(paths: tuple[Path, ...] = DEFAULT_CSV_PATHS, limit: int = 4000) -> list[dict[str, Any]]:
    rows: list[dict[str, Any]] = []
    for path in paths:
        if not path.exists():
            continue
        df = pd.read_csv(path)
        title_col = "title" if "title" in df.columns else df.columns[0]
        body_col = "text" if "text" in df.columns else ("body" if "body" in df.columns else title_col)
        cat_col = "category" if "category" in df.columns else None
        for i, row in df.head(limit).iterrows():
            title = str(row.get(title_col) or "").strip()
            body = str(row.get(body_col) or "").strip()
            if not title and not body:
                continue
            rows.append(
                {
                    "id": f"{path.stem}-{i}",
                    "title": title or body[:80],
                    "text": body or title,
                    "category": str(row.get(cat_col) or "") if cat_col else "",
                    "source_tag": path.stem,
                }
            )
    return rows


def ensure_knowledge_base_loaded(*, force: bool = False) -> int:
    """Push CSV knowledge into DB if the table is empty (or force reload)."""
    from database.connection import db_cursor, ensure_ml_training_schema

    ensure_ml_training_schema()
    with db_cursor() as cur:
        cur.execute("SELECT COUNT(*) AS n FROM knowledge_base")
        count = int((cur.fetchone() or {}).get("n") or 0)
        if count > 0 and not force:
            return count
        if force:
            cur.execute("TRUNCATE knowledge_base RESTART IDENTITY")
        docs = _docs_from_csv()
        for doc in docs:
            cur.execute(
                """
                INSERT INTO knowledge_base (title, body, category, source_tag)
                VALUES (%s, %s, %s, %s)
                """,
                (doc["title"], doc["text"], doc.get("category") or None, doc.get("source_tag")),
            )
        log.info("Loaded %s knowledge_base rows from CSV", len(docs))
        return len(docs)


def load_knowledge_documents(*, limit: int = 2000) -> list[dict[str, Any]]:
    """Load KB docs from DB; fall back to CSV files if DB unavailable/empty."""
    try:
        from database.connection import db_cursor, ensure_ml_training_schema

        ensure_ml_training_schema()
        ensure_knowledge_base_loaded()
        with db_cursor() as cur:
            cur.execute(
                """
                SELECT kb_id AS id, title, body AS text, category, source_tag
                FROM knowledge_base
                ORDER BY kb_id
                LIMIT %s
                """,
                (limit,),
            )
            rows = [dict(r) for r in (cur.fetchall() or [])]
            if rows:
                return rows
    except Exception as exc:  # noqa: BLE001
        log.warning("knowledge_base DB load failed, using CSV: %s", exc)
    return _docs_from_csv(limit=limit)


def score_content_relevancy(
    title: str,
    body: str = "",
    *,
    limit: int = 8,
    min_score: float = 0.08,
) -> dict[str, Any]:
    """Rank draft content against the newsroom knowledge base."""
    query = f"{title or ''}. {body or ''}".strip()
    docs = load_knowledge_documents()
    hits = [h for h in rank(query, docs, limit=limit) if float(h.get("score") or 0) >= min_score]
    best = float(hits[0]["score"]) if hits else 0.0
    if best >= 0.35:
        label = "high"
    elif best >= 0.15:
        label = "medium"
    elif best > 0:
        label = "low"
    else:
        label = "none"
    return {
        "relevancy": label,
        "best_score": round(best, 4),
        "matches": hits,
        "knowledge_base_size": len(docs),
        "source": "knowledge_base",
    }

# Naming guide (keep it simple)

This is a college project. Names should say **what** something is.

## Two different “priority” ideas (do not mix them)

| Name in code | Who uses it | Meaning |
|--------------|-------------|---------|
| **Editor queue urgency** | Editor / Manager | How soon should this story be reviewed? (`urgent` / `high` / `medium` / `low`) |
| **Pitch importance** | AIA story pitches | How important is this pitch idea? Simple formula. Stored in DB column `story_pitch.priority` |

- Dataset: `ml/editor/dataset/synthetic_news.csv`
- Model file: `artifacts/models/editor_queue_urgency.joblib`
- Function: `predict_editor_queue_urgency(...)`
- API field on content queue: `editor_queue_urgency`

Pitch responses use `pitch_importance` (and still write the DB column named `priority`).

## News vs article vs story pitch (do not mix)

| Thing | What it is | DB |
|-------|------------|-----|
| **News** | Event / breaking-style piece | `content` + `news` (`content_kind = news`) |
| **Article** | Feature / longer story | `content` + `article` (`content_kind = article`) |
| **Story pitch** | Idea *before* writing — meant for an **article** | `story_pitch` → later linked via `article.story_pitch_id` |

Story pitches are **not** for the `news` kind. The schema already links pitches to `article`, not to `news`.

## Flags vs topics (different things)

| Field | Where | Meaning |
|-------|-------|---------|
| `breaking_news_flag` | `news` table | Only boolean “flag” for news: is this breaking? |
| `severity` | `news` table | low / medium / high / critical (not a flag) |
| `content_kind` | `content` table | `news` or `article` |
| **topic / category** | Manual entry + `category` table | Politics, Sports, Business, … — **not flags**, **not a trained model** |

So: **Sports / Politics / Weather are categories (topics), not flags.**  
The only news boolean flag in the schema is **`breaking_news_flag`**.

## Content adaptation (CAE)

One shared model for all platforms: `artifacts/models/cae/`.  
Platform still matters at predict time (sentence limits + thresholds).

## Other clear names

| Name | Meaning |
|------|---------|
| `engagement_score` | Rough “will people interact?” score (formula) |
| `audience_interest_score` | Rough interest intensity (formula) |
| `trend_score` | Rough trendiness (formula) |
| `source_type` | `AI_GENERATED` (AIA) vs `MANUAL` |

## Layer names

| Folder | Job in one line |
|--------|-----------------|
| `api/schemas/` | JSON shapes for HTTP |
| `api/routers/` | HTTP endpoints |
| `modules/` | Thin desk façade (login user + call a service) |
| `services/` | Real business logic |
| `ml/` | Scoring + training (urgency, CAE, knowledge, AIA formulas) |
| `artifacts/models/` | Saved joblib models |

def login(client: httpx.Client, username: str) -> str:
    r = client.post(
        "/auth/login",
        json={"username": username, "password": DEMO_PASSWORDS[username]},
    )
    r.raise_for_status()
    return r.json()["session_token"]


def call(
    client: httpx.Client,
    method: str,
    path: str,
    *,
    token: str | None = None,
    json_body: dict[str, Any] | None = None,
    expect: set[int] | None = None,
) -> tuple[str, int, str]:
    headers = {"Authorization": f"Bearer {token}"} if token else {}
    r = client.request(method, path, headers=headers, json=json_body)
    ok_codes = expect or {200}
    status = "PASS" if r.status_code in ok_codes else "FAIL"
    detail = ""
    if status == "FAIL":
        detail = (r.text or "")[:180].replace("\n", " ")
    return f"{method} {path}", r.status_code, f"{status}{(' | ' + detail) if detail else ''}"


def run() -> int:
    results: list[tuple[str, str, int, str]] = []
    with httpx.Client(base_url=BASE, timeout=60.0) as client:
        # Health / docs
        for method, path in (("GET", "/docs"), ("GET", "/openapi.json")):
